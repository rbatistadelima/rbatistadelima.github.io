// system/decisions.js
export const decisions = {
    currentDecision: null,
    decisionQueue: [],
    decisionHistory: [],
    
    // Present a decision to the player
    present(decision) {
        if (this.currentDecision) {
            this.queue(decision);
            return false;
        }
        
        this.currentDecision = {
            ...decision,
            id: Date.now() + Math.random().toString(36).substr(2, 5),
            presentedAt: Date.now()
        };
        
        return true;
    },
    
    // Queue a decision for later presentation
    queue(decision) {
        this.decisionQueue.push({
            ...decision,
            id: Date.now() + Math.random().toString(36).substr(2, 5),
            queuedAt: Date.now()
        });
    },
    
    // Process the decision queue
    processQueue() {
        if (!this.currentDecision && this.decisionQueue.length > 0) {
            this.present(this.decisionQueue.shift());
            return true;
        }
        return false;
    },
    
    // Make a choice in the current decision
    choose(choiceIndex) {
        if (!this.currentDecision || 
            !this.currentDecision.choices || 
            !this.currentDecision.choices[choiceIndex]) {
            return false;
        }
        
        const choice = this.currentDecision.choices[choiceIndex];
        
        // Add to history
        this.decisionHistory.push({
            decision: this.currentDecision,
            choice: choice,
            chosenAt: Date.now()
        });
        
        // Keep history manageable
        if (this.decisionHistory.length > 50) {
            this.decisionHistory.shift();
        }
        
        // Execute choice action if defined
        if (choice.action) {
            this.executeAction(choice.action, choice.data);
        }
        
        // Execute consequence if defined
        if (choice.consequence) {
            choice.consequence();
        }
        
        // Clear current decision
        this.currentDecision = null;
        
        // Process next decision in queue
        this.processQueue();
        
        return true;
    },
    
    // Execute a decision action
    executeAction(action, data) {
        switch (action) {
            case "attack":
                // Combat system would handle this
                console.log("Attack action:", data);
                break;
                
            case "flee":
                // Chance to flee from combat
                console.log("Flee action:", data);
                break;
                
            case "take_item":
                // Add item to inventory
                console.log("Take item action:", data);
                break;
                
            case "move":
                // Move to new location
                console.log("Move action:", data);
                break;
                
            case "explore":
                // Explore current location
                console.log("Explore action:", data);
                break;
                
            case "rest":
                // Rest and recover health
                console.log("Rest action:", data);
                break;
                
            default:
                console.log("Unknown action:", action, data);
        }
    },

    // Check if there's an active decision
    hasActiveDecision() {
        return this.currentDecision !== null;
    },
    
    // Get the current decision
    getCurrentDecision() {
        return this.currentDecision;
    },
    
    // Get decision history
    getHistory() {
        return [...this.decisionHistory];
    },
    
    // Clear all decisions
    clearAll() {
        this.currentDecision = null;
        this.decisionQueue = [];
    },
    
    // Get save data for decisions
    getSaveData() {
        return {
            decisionQueue: JSON.parse(JSON.stringify(this.decisionQueue)),
            decisionHistory: JSON.parse(JSON.stringify(this.decisionHistory))
        };
    },
    
    // Load decisions from save data
    loadSaveData(data) {
        this.decisionQueue = data.decisionQueue || [];
        this.decisionHistory = data.decisionHistory || [];
    }
};
