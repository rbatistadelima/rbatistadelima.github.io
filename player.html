// system/player.js
export const player = {
    name: "Adventurer",
    level: 1,
    exp: 0,
    expToNext: 100,
    health: 100,
    maxHealth: 100,
    gold: 50,
    stats: {
        str: 10,
        dex: 8,
        int: 6,
        vit: 12
    },
    location: "forest",
    statusEffects: [],
    
    // Add experience points and check for level up
    addExp(amount) {
        this.exp += amount;
        if (this.exp >= this.expToNext) {
            return this.levelUp();
        }
        return false;
    },
    
    // Handle level up process
    levelUp() {
        this.level++;
        this.exp -= this.expToNext;
        this.expToNext = Math.floor(this.expToNext * 1.5);
        this.maxHealth += 10;
        this.health = this.maxHealth;
        
        // Random stat increases
        this.stats.str += Math.floor(Math.random() * 2) + 1;
        this.stats.dex += Math.floor(Math.random() * 2) + 1;
        this.stats.int += Math.floor(Math.random() * 2) + 1;
        this.stats.vit += Math.floor(Math.random() * 2) + 1;
        
        return true;
    },
    
    // Apply damage to player and check for death
    takeDamage(amount) {
        this.health -= amount;
        return this.health <= 0;
    },
    
    // Heal the player
    heal(amount) {
        this.health = Math.min(this.health + amount, this.maxHealth);
    },
    
    // Add a status effect
    addStatusEffect(effect) {
        this.statusEffects.push({
            ...effect,
            id: Date.now() + Math.random().toString(36).substr(2, 5)
        });
    },
    
    // Remove a status effect
    removeStatusEffect(effectId) {
        this.statusEffects = this.statusEffects.filter(effect => effect.id !== effectId);
    },
    
    // Update status effects (called regularly)
    updateStatusEffects() {
        this.statusEffects = this.statusEffects.filter(effect => {
            effect.duration--;
            return effect.duration > 0;
        });
    },
    
    // Calculate total stats (base + equipment + status effects)
    getTotalStats() {
        // This would incorporate equipment and status effects in a real implementation
        return {...this.stats};
    },
    
    // Move to a new location
    moveTo(location) {
        this.location = location;
        return true;
    },
    
    // Get player data for saving
    getSaveData() {
        return {
            name: this.name,
            level: this.level,
            exp: this.exp,
            expToNext: this.expToNext,
            health: this.health,
            maxHealth: this.maxHealth,
            gold: this.gold,
            stats: {...this.stats},
            location: this.location,
            statusEffects: [...this.statusEffects]
        };
    },
    
    // Load player data
    loadSaveData(data) {
        this.name = data.name;
        this.level = data.level;
        this.exp = data.exp;
        this.expToNext = data.expToNext;
        this.health = data.health;
        this.maxHealth = data.maxHealth;
        this.gold = data.gold;
        this.stats = {...data.stats};
        this.location = data.location;
        this.statusEffects = [...data.statusEffects];
    },
    
    // Reset player to initial state
    reset() {
        this.name = "Adventurer";
        this.level = 1;
        this.exp = 0;
        this.expToNext = 100;
        this.health = 100;
        this.maxHealth = 100;
        this.gold = 50;
        this.stats = {
            str: 10,
            dex: 8,
            int: 6,
            vit: 12
        };
        this.location = "forest";
        this.statusEffects = [];
    }
};

This module handles all player-related data and functionality including stats, leveling, health, and status effects.
