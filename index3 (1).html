Player System (player.js)

// system/player.js
export const player = {
    name: "Adventurer",
    level: 1,
    exp: 0,
    expToNext: 100,
    health: 100,
    maxHealth: 100,
    gold: 50,
    stats: {
        str: 10,
        dex: 8,
        int: 6,
        vit: 12
    },
    location: "forest",
    
    addExp(amount) {
        this.exp += amount;
        if (this.exp >= this.expToNext) {
            this.levelUp();
        }
    },
    
    levelUp() {
        this.level++;
        this.exp -= this.expToNext;
        this.expToNext = Math.floor(this.expToNext * 1.5);
        this.maxHealth += 10;
        this.health = this.maxHealth;
        
        // Random stat increases
        this.stats.str += Math.floor(Math.random() * 2) + 1;
        this.stats.dex += Math.floor(Math.random() * 2) + 1;
        this.stats.int += Math.floor(Math.random() * 2) + 1;
        this.stats.vit += Math.floor(Math.random() * 2) + 1;
        
        return true;
    },
    
    takeDamage(amount) {
        this.health -= amount;
        return this.health <= 0;
    },
    
    heal(amount) {
        this.health = Math.min(this.health + amount, this.maxHealth);
    }
};

Inventory System (inventory.js)

// system/inventory.js
export const inventory = {
    items: [],
    maxSize: 20,
    equipment: {
        weapon: null,
        armor: null,
        accessory: null
    },
    
    addItem(item) {
        if (this.items.length >= this.maxSize) return false;
        
        const existingItem = this.items.find(i => i.id === item.id);
        if (existingItem && existingItem.stackable) {
            existingItem.quantity += item.quantity || 1;
        } else {
            this.items.push({...item, quantity: item.quantity || 1});
        }
        return true;
    },
    
    removeItem(itemId, quantity = 1) {
        const index = this.items.findIndex(i => i.id === itemId);
        if (index === -1) return false;
        
        if (this.items[index].stackable && this.items[index].quantity > quantity) {
            this.items[index].quantity -= quantity;
        } else {
            this.items.splice(index, 1);
        }
        return true;
    },
    
    equipItem(itemId) {
        const item = this.items.find(i => i.id === itemId);
        if (!item || !item.equippable) return false;
        
        // Unequip current item of this type if needed
        if (this.equipment[item.slot]) {
            this.unequipItem(item.slot);
        }
        
        this.equipment[item.slot] = item;
        this.removeItem(itemId);
        return true;
    },
    
    unequipItem(slot) {
        if (!this.equipment[slot]) return false;
        
        this.addItem(this.equipment[slot]);
        this.equipment[slot] = null;
        return true;
    },
    
    getStats() {
        let stats = { str: 0, dex: 0, int: 0, vit: 0 };
        for (const slot in this.equipment) {
            if (this.equipment[slot] && this.equipment[slot].stats) {
                for (const stat in this.equipment[slot].stats) {
                    stats[stat] += this.equipment[slot].stats[stat];
                }
            }
        }
        return stats;
    }
};

// system/events.js
export const events = {
    eventQueue: [],
    eventHandlers: {},
    
    trigger(eventName, data = {}) {
        this.eventQueue.push({ name: eventName, data, timestamp: Date.now() });
        
        if (this.eventHandlers[eventName]) {
            this.eventHandlers[eventName].forEach(handler => handler(data));
        }
    },
    
    on(eventName, handler) {
        if (!this.eventHandlers[eventName]) {
            this.eventHandlers[eventName] = [];
        }
        this.eventHandlers[eventName].push(handler);
    },
    
    processQueue() {
        if (this.eventQueue.length > 0) {
            const event = this.eventQueue.shift();
            // Process event here
            return event;
        }
        return null;
    },
    
    // Common game events
    combatEvents: {

Progress System (progress.js)

// system/progress.js
export const progress = {
    resources: {
        wood: 0,
        stone: 0,
        iron: 0,
        gold: 0
    },
    
    jobs: {
        lumberjack: {
            name: "Lumberjack",
            level: 0,
            rate: 0.1,  // resources per second
            resource: "wood",
            cost: { wood: 10, gold: 5 }
        },
        miner: {
            name: "Miner",
            level: 0,
            rate: 0.05,
            resource: "stone",
            cost: { wood: 5, stone: 10, gold: 8 }
        }
    },
    
    upgrades: {
        better_tools: {
            name: "Better Tools",
            description: "Increases resource gathering rate by 20%",
            cost: { wood: 50, stone: 25, gold: 20 },
            effect: (job) => job.rate *= 1.2,
            purchased: false
        }
    },
    
    update(deltaTime) {
        // Update resource generation based on jobs
        for (const jobId in this.jobs) {
            const job = this.jobs[jobId];
            if (job.level > 0) {
                this.resources[job.resource] += job.rate * job.level * deltaTime;
            }
        }
    },
    
    upgradeJob(jobId) {
        const job = this.jobs[jobId];
        if (!job) return false;
        
        // Check if player can afford the upgrade
        for (const resource in job.cost) {
            if (this.resources[resource] < job.cost[resource] * (job.level + 1)) {
                return false;
            }
        }
        
        // Deduct costs
        for (const resource in job.cost) {
            this.resources[resource] -= job.cost[resource] * (job.level + 1);
        }
        
        job.level++;
        return true;
    },
    
    purchaseUpgrade(upgradeId) {
        const upgrade = this.upgrades[upgradeId];
        if (!upgrade || upgrade.purchased) return false;
        
        // Check if player can afford the upgrade
        for (const resource in upgrade.cost) {
            if (this.resources[resource] < upgrade.cost[resource]) {
                return false;
            }
        }
        
        // Deduct costs
        for (const resource in upgrade.cost) {
            this.resources[resource] -= upgrade.cost[resource];
        }
        
        // Apply upgrade effect
        if (upgrade.effect) {
            for (const jobId in this.jobs) {
                upgrade.effect(this.jobs[jobId]);
            }
        }
        
        upgrade.purchased = true;
        return true;
    },
    
    getSaveData() {
        return {
            resources: {...this.resources},
            jobs: JSON.parse(JSON.stringify(this.jobs)),
            upgrades: JSON.parse(JSON.stringify(this.upgrades))
        };
    },
    
    loadSaveData(data) {
        this.resources = {...data.resources};
        this.jobs = data.jobs;
        this.upgrades = data.upgrades;
    }
};

Save System (save-system.js)

// system/save-system.js
export const saveSystem = {
    saveKey: "idleAsciiRPG_saveData",
    
    save() {
        const saveData = {
            player: player,
            inventory: inventory,
            progress: progress.getSaveData(),
            timestamp: Date.now(),
            version: "1.0"
        };
        
        try {
            localStorage.setItem(this.saveKey, JSON.stringify(saveData));
            return true;
        } catch (e) {
            console.error("Save failed:", e);
            return false;
        }
    },
    
    load() {
        try {
            const saveData = JSON.parse(localStorage.getItem(this.saveKey));
            if (!saveData) return false;
            
            // Validate save data
            if (!saveData.version || !saveData.player) return false;
            
            // Load data into systems
            Object.assign(player, saveData.player);
            Object.assign(inventory, saveData.inventory);
            progress.loadSaveData(saveData.progress);
            
            return true;
        } catch (e) {
            console.error("Load failed:", e);
            return false;
        }
    },
    
    deleteSave() {
        try {
            localStorage.removeItem(this.saveKey);
            return true;
        } catch (e) {
            console.error("Delete save failed:", e);
            return false;
        }
    },
    
    exportSave() {
        try {
            const saveData = localStorage.getItem(this.saveKey);
            return btoa(unescape(encodeURIComponent(saveData)));
        } catch (e) {
            console.error("Export failed:", e);
            return null;
        }
    },
    
    importSave(encodedData) {
        try {
            const saveData = decodeURIComponent(escape(atob(encodedData)));
            localStorage.setItem(this.saveKey, saveData);
            return this.load();
        } catch (e) {
            console.error("Import failed:", e);
            return false;
        }
    },
    
    autoSave(interval = 30000) {
        setInterval(() => {
            if (this.save()) {
                events.trigger("autosave_complete");
            }
        }, interval);
    }
};
