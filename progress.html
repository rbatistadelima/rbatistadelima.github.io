// system/progress.js
export const progress = {
    resources: {
        wood: 0,
        stone: 0,
        iron: 0,
        gold: 0,
        essence: 0
    },
    
    jobs: {
        lumberjack: {
            name: "Lumberjack",
            level: 0,
            rate: 0.1,
            resource: "wood",
            cost: { wood: 10, gold: 5 },
            unlocked: true
        },
        miner: {
            name: "Miner",
            level: 0,
            rate: 0.05,
            resource: "stone",
            cost: { wood: 5, stone: 10, gold: 8 },
            unlocked: false
        },
        blacksmith: {
            name: "Blacksmith",
            level: 0,
            rate: 0.03,
            resource: "iron",
            cost: { wood: 20, stone: 15, gold: 15 },
            unlocked: false
        },
        alchemist: {
            name: "Alchemist",
            level: 0,
            rate: 0.02,
            resource: "essence",
            cost: { wood: 10, stone: 10, iron: 5, gold: 20 },
            unlocked: false
        }
    },
    
    upgrades: {
        better_tools: {
            name: "Better Tools",
            description: "Increases resource gathering rate by 20%",
            cost: { wood: 50, stone: 25, gold: 20 },
            effect: (job) => job.rate *= 1.2,
            purchased: false
        },
        efficiency: {
            name: "Efficiency Training",
            description: "Reduces job upgrade costs by 15%",
            cost: { wood: 100, stone: 50, gold: 40 },
            effect: () => {
                // This would be applied during cost calculation
            },
            purchased: false
        },
        automation: {
            name: "Basic Automation",
            description: "Jobs continue working when offline",
            cost: { wood: 200, stone: 100, gold: 100 },
            effect: () => {
                // Enable offline progress
            },
            purchased: false
        }
    },
    
    achievements: {
        first_tree: {
            name: "First Tree",
            description: "Chop down your first tree",
            completed: false,
            reward: { gold: 10 }
        },
        stone_age: {
            name: "Stone Age",
            description: "Gather 100 stone",
            completed: false,
            reward: { stone: 50 }
        },
        master_lumberjack: {
            name: "Master Lumberjack",
            description: "Reach level 10 Lumberjack",
            completed: false,
            reward: { wood: 200, gold: 50 }
        }
    },
    
    // Update progress based on elapsed time
    update(deltaTime) {
        // Update resource generation based on jobs
        for (const jobId in this.jobs) {
            const job = this.jobs[jobId];
            if (job.level > 0 && job.unlocked) {
                this.resources[job.resource] += job.rate * job.level * deltaTime;
                
                // Check achievements
                this.checkAchievements();
            }
        }
    },
    
    // Upgrade a job
    upgradeJob(jobId) {
        const job = this.jobs[jobId];
        if (!job || !job.unlocked) return false;
        
        // Calculate cost with potential discounts
        const costMultiplier = 1.0; // Would be reduced by upgrades
        const actualCost = {};
        for (const resource in job.cost) {
            actualCost[resource] = job.cost[resource] * (job.level + 1) * costMultiplier;
            if (this.resources[resource] < actualCost[resource]) {
                return false;
            }
        }
        
        // Deduct costs
        for (const resource in actualCost) {
            this.resources[resource] -= actualCost[resource];
        }
        
        job.level++;
        return true;
    },
    
    // Unlock a job
    unlockJob(jobId) {
        const job = this.jobs[jobId];
        if (!job || job.unlocked) return false;
        
        // Check requirements (simplified)
        const requirements = {
            miner: { wood: 50 },
            blacksmith: { wood: 100, stone: 50 },
            alchemist: { wood: 150, stone: 100, iron: 25 }
        };
        
        if (requirements[jobId]) {
            for (const resource in requirements[jobId]) {
                if (this.resources[resource] < requirements[jobId][resource]) {
                    return false;
                }
            }
        }
        
        job.unlocked = true;
        return true;
    },
    
    // Purchase an upgrade
    purchaseUpgrade(upgradeId) {
        const upgrade = this.upgrades[upgradeId];
        if (!upgrade || upgrade.purchased) return false;
        
        // Check if player can afford the upgrade
        for (const resource in upgrade.cost) {
            if (this.resources[resource] < upgrade.cost[resource]) {
                return false;
            }
        }
        
        // Deduct costs
        for (const resource in upgrade.cost) {
            this.resources[resource] -= upgrade.cost[resource];
        }
        
        // Apply upgrade effect
        if (upgrade.effect) {
            for (const jobId in this.jobs) {
                upgrade.effect(this.jobs[jobId]);
            }
        }
        
        upgrade.purchased = true;
        return true;
    },
    
    // Check and complete achievements
    checkAchievements() {
        // First Tree
        if (!this.achievements.first_tree.completed && this.resources.wood >= 1) {
            this.achievements.first_tree.completed = true;
            this.applyReward(this.achievements.first_tree.reward);
        }
        
        // Stone Age
        if (!this.achievements.stone_age.completed && this.resources.stone >= 100) {
            this.achievements.stone_age.completed = true;
            this.applyReward(this.achievements.stone_age.reward);
        }
        
        // Master Lumberjack
        if (!this.achievements.master_lumberjack.completed && 
            this.jobs.lumberjack.level >= 10) {
            this.achievements.master_lumberjack.completed = true;
            this.applyReward(this.achievements.master_lumberjack.reward);
        }
    },
    
    // Apply achievement rewards
    applyReward(reward) {
        for (const resource in reward) {
            this.resources[resource] += reward[resource];
        }
    },
    
    // Get total production rates
    getProductionRates() {
        const rates = {};
        for (const resource in this.resources) {
            rates[resource] = 0;
        }
        
        for (const jobId in this.jobs) {
            const job = this.jobs[jobId];
            if (job.level > 0 && job.unlocked) {
                rates[job.resource] += job.rate * job.level;
            }
        }
        
        return rates;
    },
    
    // Get save data for progress
    getSaveData() {
        return {
            resources: {...this.resources},
            jobs: JSON.parse(JSON.stringify(this.jobs)),
            upgrades: JSON.parse(JSON.stringify(this.upgrades)),
            achievements: JSON.parse(JSON.stringify(this.achievements))
        };
    },
    
    // Load progress from save data
    loadSaveData(data) {
        this.resources = {...data.resources};
        this.jobs = data.jobs;
        this.upgrades = data.upgrades;
        this.achievements = data.achievements;
    },
    
    // Reset progress to initial state
    reset() {
        this.resources = {
            wood: 0,
            stone: 0,
            iron: 0,
            gold: 0,
            essence: 0
        };
        
        for (const jobId in this.jobs) {
            this.jobs[jobId].level = 0;
            this.jobs[jobId].unlocked = jobId === "lumberjack";
        }
        
        for (const upgradeId in this.upgrades) {
            this.upgrades[upgradeId].purchased = false;
        }
        
        for (const achievementId in this.achievements) {
            this.achievements[achievementId].completed = false;
        }
    }
};
