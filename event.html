Event System (events.js)

// system/events.js
export const events = {
    eventQueue: [],
    eventHandlers: {},
    lastEventTime: 0,
    eventCooldown: 1000, // ms between events
    
    // Trigger a new event
    trigger(eventName, data = {}) {
        const event = {
            name: eventName,
            data: data,
            timestamp: Date.now(),
            id: Date.now() + Math.random().toString(36).substr(2, 5)
        };
        
        this.eventQueue.push(event);
        
        // Immediately notify handlers if any are registered
        if (this.eventHandlers[eventName]) {
            this.eventHandlers[eventName].forEach(handler => handler(event.data));
        }
        
        return event.id;
    },
    
    // Register an event handler
    on(eventName, handler) {
        if (!this.eventHandlers[eventName]) {
            this.eventHandlers[eventName] = [];
        }
        this.eventHandlers[eventName].push(handler);
    },
    
    // Remove an event handler
    off(eventName, handler) {
        if (!this.eventHandlers[eventName]) return false;
        
        const index = this.eventHandlers[eventName].indexOf(handler);
        if (index !== -1) {
            this.eventHandlers[eventName].splice(index, 1);
            return true;
        }
        return false;
    },
    
    // Process the next event in the queue
    processNext() {
        if (this.eventQueue.length === 0) return null;
        
        const now = Date.now();
        if (now - this.lastEventTime < this.eventCooldown) return null;
        
        this.lastEventTime = now;
        return this.eventQueue.shift();
    },
    
    // Process all available events
    processAll() {
        const processed = [];
        let event;
        
        while ((event = this.processNext()) !== null) {
            processed.push(event);
        }
        
        return processed;
    },
    
    // Clear all events from the queue
    clearQueue() {
        this.eventQueue = [];
    },
    
    // Get the number of events in the queue
    getQueueLength() {
        return this.eventQueue.length;
    },
    
    // Check if an event type has handlers
    hasHandlers(eventName) {
        return this.eventHandlers[eventName] && this.eventHandlers[eventName].length > 0;
    },
    
    // Predefined game events
    combat: {
        encounter: (enemyType, difficulty = 1) => {
            return {
                type: "combat",
                enemy: enemyType,
                difficulty: difficulty,
                message: `A ${enemyType} appears!`
            };
        },
        
        victory: (enemy, rewards = {}) => {
            return {
                type: "combat_result",
                result: "victory",
                enemy: enemy,
                rewards: rewards,
                message: `You defeated the ${enemy}!`
            };
        },
        
        defeat: (enemy, penalty = {}) => {
            return {
                type: "combat_result",
                result: "defeat",
                enemy: enemy,
                penalty: penalty,
                message: `You were defeated by the ${enemy}!`
            };
        }
    },
    
    exploration: {
        discover: (location, items = []) => {
            return {
                type: "discovery",
                location: location,
                items: items,
                message: `You discovered ${location}!`
            };
        },
        
        findItem: (item, quantity = 1) => {
            return {
                type: "item_found",
                item: item,
                quantity: quantity,
                message: `You found ${quantity > 1 ? quantity + ' ' : ''}${item}!`
            };
        },
        
        trap: (damage = 10) => {
            return {
                type: "trap",
                damage: damage,
                message: "You triggered a trap!"
            };
        }
    },
    
    player: {
        levelUp: (newLevel, stats = {}) => {
            return {
                type: "level_up",
                level: newLevel,
                stats: stats,
                message: `You reached level ${newLevel}!`
            };
        },
        
        statIncrease: (stat, amount = 1) => {
            return {
                type: "stat_increase",
                stat: stat,
                amount: amount,
                message: `Your ${stat} increased by ${amount}!`
            };
        },
        
        damage: (amount, source = "unknown") => {
            return {
                type: "damage",
                amount: amount,
                source: source,
                message: `You took ${amount} damage from ${source}!`
            };
        },
        
        heal: (amount, source = "unknown") => {
            return {
                type: "heal",
                amount: amount,
                source: source,
                message: `You recovered ${amount} HP from ${source}!`
            };
        }
    },
    
    world: {
        timeChange: (newTime) => {
            return {
                type: "time_change",
                time: newTime,
                message: `Time changes to ${newTime}`
            };
        },
        
        weatherChange: (newWeather) => {
            return {
                type: "weather_change",
                weather: newWeather,
                message: `Weather changes to ${newWeather}`
            };
        }
    },
    
    // Get save data for events
    getSaveData() {
        return {
            eventQueue: JSON.parse(JSON.stringify(this.eventQueue)),
            eventHandlers: {} // Handlers are functions, so we don't save them
        };
    },
    
    // Load events from save data
    loadSaveData(data) {
        this.eventQueue = data.eventQueue || [];
        // Handlers are re-registered by game systems during initialization
    }
};

This module handles all game events, event queuing, and event processing with a handler system.
