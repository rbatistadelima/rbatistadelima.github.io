Inventory System (inventory.js)

// system/inventory.js
export const inventory = {
    items: [],
    maxSize: 20,
    equipment: {
        weapon: null,
        armor: null,
        accessory: null
    },
    
    // Item database - predefined items in the game
    itemDB: {
        health_potion: {
            id: "health_potion",
            name: "Health Potion",
            type: "consumable",
            stackable: true,
            maxStack: 10,
            use: (player) => {
                player.heal(25);
                return "You drink a health potion and recover 25 HP.";
            }
        },
        mana_potion: {
            id: "mana_potion",
            name: "Mana Potion",
            type: "consumable",
            stackable: true,
            maxStack: 10
        },
        rusty_sword: {
            id: "rusty_sword",
            name: "Rusty Sword",
            type: "weapon",
            equippable: true,
            slot: "weapon",
            stats: { str: 3, dex: 1 }
        },
        iron_sword: {
            id: "iron_sword",
            name: "Iron Sword",
            type: "weapon",
            equippable: true,
            slot: "weapon",
            stats: { str: 5, dex: 2 }
        },
        leather_armor: {
            id: "leather_armor",
            name: "Leather Armor",
            type: "armor",
            equippable: true,
            slot: "armor",
            stats: { vit: 4, dex: 1 }
        },
        amulet: {
            id: "amulet",
            name: "Bronze Amulet",
            type: "accessory",
            equippable: true,
            slot: "accessory",
            stats: { int: 2, vit: 1 }
        }
    },
    
    // Add an item to inventory
    addItem(itemId, quantity = 1) {
        const itemTemplate = this.itemDB[itemId];
        if (!itemTemplate) return false;
        
        // Check if we can add to existing stack
        if (itemTemplate.stackable) {
            const existingItem = this.items.find(i => i.id === itemId && i.quantity < i.maxStack);
            if (existingItem) {
                const spaceLeft = existingItem.maxStack - existingItem.quantity;
                const addAmount = Math.min(quantity, spaceLeft);
                existingItem.quantity += addAmount;
                
                if (addAmount < quantity) {
                    // Create new stack with remaining items
                    const newStack = { ...itemTemplate, quantity: quantity - addAmount };
                    if (this.items.length < this.maxSize) {
                        this.items.push(newStack);
                        return true;
                    }
                    return false;
                }
                return true;
            }
        }
        
        // Create new item stack
        if (this.items.length >= this.maxSize) return false;
        
        const newItem = { ...itemTemplate, quantity: quantity };
        this.items.push(newItem);
        return true;
    },
    
    // Remove an item from inventory
    removeItem(itemId, quantity = 1) {
        const itemIndex = this.items.findIndex(i => i.id === itemId);
        if (itemIndex === -1) return false;
        
        const item = this.items[itemIndex];
        if (item.stackable && item.quantity > quantity) {
            item.quantity -= quantity;
        } else {
            this.items.splice(itemIndex, 1);
        }
        return true;
    },
    
    // Use a consumable item
    useItem(itemId, player) {
        const item = this.items.find(i => i.id === itemId);
        if (!item || item.type !== "consumable") return null;
        
        const result = item.use ? item.use(player) : `You use the ${item.name}.`;
        this.removeItem(itemId, 1);
        return result;
    },
    
    // Equip an item
    equipItem(itemId) {
        const itemIndex = this.items.findIndex(i => i.id === itemId);
        if (itemIndex === -1 || !this.items[itemIndex].equippable) return false;
        
        const item = this.items[itemIndex];
        const slot = item.slot;
        
        // Unequip current item if needed
        if (this.equipment[slot]) {
            this.unequipItem(slot);
        }
        
        // Equip the new item
        this.equipment[slot] = { ...item };
        this.items.splice(itemIndex, 1);
        return true;
    },
    
    // Unequip an item
    unequipItem(slot) {
        if (!this.equipment[slot]) return false;
        
        if (this.items.length >= this.maxSize) return false;
        
        this.items.push({ ...this.equipment[slot] });
        this.equipment[slot] = null;
        return true;
    },
    
    // Get total stats from equipped items
    getEquipmentStats() {
        let stats = { str: 0, dex: 0, int: 0, vit: 0 };
        
        for (const slot in this.equipment) {
            const item = this.equipment[slot];
            if (item && item.stats) {
                for (const stat in item.stats) {
                    stats[stat] += item.stats[stat];
                }
            }
        }
        
        return stats;
    },
    
    // Check if player has an item
    hasItem(itemId, quantity = 1) {
        const item = this.items.find(i => i.id === itemId);
        if (!item) return false;
        return item.quantity >= quantity;
    },
    
    // Get save data for inventory
    getSaveData() {
        return {
            items: JSON.parse(JSON.stringify(this.items)),
            equipment: JSON.parse(JSON.stringify(this.equipment))
        };
    },
    
    // Load inventory from save data
    loadSaveData(data) {
        this.items = data.items || [];
        this.equipment = data.equipment || {
            weapon: null,
            armor: null,
            accessory: null
        };
    },
    
    // Initialize with default items
    initialize() {
        this.addItem("health_potion", 3);
        this.addItem("rusty_sword");
    }
};

This module handles all inventory management, items, equipment, and item usage.
